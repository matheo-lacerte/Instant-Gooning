export async function getAllGames(req, res) {
    try {
        const { data, error } = await supabase
            .from("games")
            .select("id, title, price, cover_url, discount, discounted_price, is_active");
        if (error) throw error;
        res.json(data);
    } catch (err) {
        res.status(500).json({ error: err.message });
    }
}
