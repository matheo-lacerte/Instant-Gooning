export async function getGameById(req, res) {
    const { id } = req.params;
    try {
        const { data, error } = await supabase
            .from("games")
            .select("*")
            .eq("id", id)
            .single();
        if (error) throw error;
        res.json(data);
    } catch (err) {
        res.status(500).json({ error: err.message });
    }
}
